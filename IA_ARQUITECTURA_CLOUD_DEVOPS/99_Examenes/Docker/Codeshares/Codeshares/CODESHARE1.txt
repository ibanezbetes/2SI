terraform {
  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 3.0"
    }
  }
}

# ⚠️ Aquí van tus credenciales del Learner Lab
provider "aws" {
  region     = "us-east-1"
  access_key = "XXX"     # Pega aquí tu AWS_ACCESS_KEY_ID
  secret_key = "X"     # Pega aquí tu AWS_SECRET_ACCESS_KEY
  token      = "XXXXXX"     # Pega aquí tu AWS_SESSION_TOKEN (es largo)
}


# Security Group para permitir SSH (22) y HTTP (80)
resource "aws_security_group" "terraform_sg" {
  name        = "terraform-sg"
  description = "Permitir SSH y HTTP"

  ingress {
    description = "SSH"
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  ingress {
    description = "HTTP"
    from_port   = 80
    to_port     = 80
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }

  tags = {
    Name = "terraform-sg"
  }
}

# Instancia EC2 Ubuntu
resource "aws_instance" "example" {
  ami                         = "ami-011899242bb902164" # Ubuntu 20.04 en us-east-1
  instance_type               = "t2.micro"
  key_name                    = "terraform-pokeapi"               # Nombre del par de claves del Learner Lab
  vpc_security_group_ids      = [aws_security_group.terraform_sg.id]
  associate_public_ip_address = true

  tags = {
    Name = "terraform-academy-test"
  }

  # Opcional: instala Nginx al arrancar
  user_data = <<-EOF
              #!/bin/bash
              apt-get update -y
              apt-get install -y nginx
              systemctl enable nginx
              systemctl start nginx
              echo "Hola desde Terraform + AWS Academy" > /var/www/html/index.nginx-debian.html
              EOF
}

# Mostrar IP pública al terminar
output "public_ip" {
  value = aws_instance.example.public_ip
}
