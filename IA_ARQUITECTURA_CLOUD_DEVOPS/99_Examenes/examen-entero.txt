import { Module } from '@nestjs/common';
import { ConfigModule, ConfigType } from '@nestjs/config';
import config from '../config/config';
import { TypeOrmModule } from '@nestjs/typeorm';

@Module({
    imports: [
        ConfigModule.forRoot({
            isGlobal: true,
            load:[config],
            envFilePath:
            process.env.NODE_ENV === 'prod'
                ? '.prod.env'
                : process.env.NODE_ENV === 'stg'
                ? '.stage.env'
                : '.env',
        }),

        TypeOrmModule.forRootAsync({
            inject: [config.KEY],
            useFactory: (cfg: ConfigType<typeof config>) => {
                const { sqliteDb } = cfg as any;

                return {
                    type: 'sqlite',
                    database: sqliteDb,
                    synchronize: true,
                    autoLoadEntities: true,
                };
            },
        }),
    ],
    providers: [],
    exports: [],
})

export class DatabaseModule {}

--- ARCHIVO: src/peliculas/peliculas.controller.ts ---

import { Body, Controller, Get, Post } from '@nestjs/common';
import { PeliculasService } from './peliculas.service';
import { CreatePeliculaDto } from './dtos/create-pelicula.dto';

@Controller('peliculas')
export class PeliculasController {

    constructor(private readonly peliculasService: PeliculasService) {}

    @Get()
    findAll() {
        return this.peliculasService.findAll();
    }

    @Post()
    create(@Body() createPeliculaDto: CreatePeliculaDto) {
        return this.peliculasService.create(createPeliculaDto);
    }
}


--- ARCHIVO: src/peliculas/peliculas.service.ts ---

import { Injectable } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Pelicula } from './entities/pelicula.entity';
import { Repository } from 'typeorm/browser/repository/Repository.js';
import { CreatePeliculaDto } from './dtos/create-pelicula.dto';

@Injectable()
export class PeliculasService {

    constructor(
        @InjectRepository(Pelicula)
        private readonly peliculaRepo: Repository<Pelicula>,
    ) { }

    findAll() {
        return this.peliculaRepo.find({order: {id: 'ASC'}});
    }

    async create(createPeliculaDto: CreatePeliculaDto) {
        const nuevaPelicula = this.peliculaRepo.create(createPeliculaDto);
        return await this.peliculaRepo.save(nuevaPelicula);
    }
}


--- ARCHIVO: src/peliculas/entities/pelicula.entity.ts ---

import { Entity, Column, PrimaryGeneratedColumn } from 'typeorm';

@Entity({name: 'peliculas1'})
export class Pelicula {
    @PrimaryGeneratedColumn()
    id: number;
    
    @Column({ type: 'varchar', length: 255, name: 'titulo' })
    titulo: string;
    
    @Column({ type: 'int', name: 'anio' })
    anio: number;
    
    @Column({ type: 'varchar', length: 255, name: 'descripcion', nullable: true })
    descripcion?: string;
    
    @Column({ type: 'varchar', length: 255, name: 'director' })
    director: string;
    
    @Column({ type: 'datetime', default: () => 'CURRENT_TIMESTAMP', name: 'created_at' })
    createdAt: Date;
}


--- ARCHIVO: src/peliculas/dtos/create-pelicula.dto.ts ---

import {
    IsString,
    IsNumber,
    IsOptional,
    IsNotEmpty,
    MaxLength,
    Min,
    Max,
} from 'class-validator';
import { Type } from 'class-transformer';

export class CreatePeliculaDto {
    @IsString()
    @IsNotEmpty({ message: 'El título es obligatorio' })
    @MaxLength(100, { message: 'El titulo no puede superar los 100 caracteres' })
    readonly titulo: string;

    @Type(() => Number)
    @IsNumber({}, { message: 'El año debe ser un número' })
    @Min(1888, { message: 'El cine nació en 1888' })
    @Max(new Date().getFullYear() + 1, {
        message: 'El año no puede ser mayor que el actual + 1',
    })
    readonly anio: number;

    @IsOptional()
    @IsString()
    readonly descripcion?: string;

    @IsOptional()
    @IsString()
    readonly director?: string;
}

import { useEffect, useState } from "react";
import { FlatList, Text, View } from "react-native";

//formatear cosas y meter elementos html

type Pelicula={
  name:string;
  url:string;
}

//array de peliculas
type PeliculaResponse={
  results:Pelicula[];
}



export default function Index() {

  //especie de getters and setters vitaminado directamente actualiza pantalla movil
  const[peliculas,setPeliculas]=useState<Pelicula[]>([]);

  //llamas al fetch, permite trabajar en segundo plano, maneja todo
  useEffect(()=>{
    const fetchPeliculas=async()=>{
      try{
        const response=await fetch("http://localhost:3000/peliculas");
        const data: PeliculaResponse =await response.json();
        console.log("Peliculas cargadas:",data.results);
        setPeliculas(data.results);  //insertamos las peliculas en el estado
      }catch(error){
        console.error("Error al cargar Peliculas:",error);
      }
    };
    fetchPeliculas();
  },[]);

  return (
    <View
      style={{   // componente visual tipo lista flatlist
        flex: 1,
        padding: 20,
      }}
    >
      <Text style={{ fontSize: 22 , marginBottom: 10}}> Listado de Peliculas </Text>
      
      {/* flatlist */}

      <FlatList
        data={peliculas}
        keyExtractor={(item) => item.name}   //extractor de nombres
        renderItem={({ item }) => (
          <Text style={{ fontSize: 18, padding: 6 }}>{item.name}</Text>
        )}
      />

    </View>
  );
}

6. SOLUCIÓN DE PROBLEMAS COMUNES
========================================

ERROR: "Connection was refused by the server"
→ El backend no está corriendo. Ejecuta: npm run start:dev

ERROR: "Cannot find module sqlite3"
→ Instala: npm install sqlite3 --legacy-peer-deps

ERROR: "Table peliculas1 does not exist"
→ Verifica que synchronize: true esté en database.module.ts
→ Reinicia el servidor: Ctrl+C y npm run start:dev

ERROR: Frontend no muestra películas
→ Cambia localhost por tu IP local (192.168.x.x)
→ Verifica en consola (F12) si hay errores de fetch

ERROR: "fatal: not a git repository"
→ Ejecuta: git init (en la raíz del proyecto)
